   157→def output_parser(name, properties, pos, uid_val=None):
   158→    props = []
   159→    for p in properties:
   160→        props.append({
   161→            "name": p["name"],
   162→            "description": p["description"],
   163→            "type": p.get("type", "string")
   164→        })
   165→    return {
   166→        "parameters": {
   167→            "schema": {
   168→                "type": "object",
   169→                "properties": {p["name"]: {"type": p.get("type", "string"), "description": p["description"]} for p in props}
   170→            }
   171→        },
   172→        "type": "@n8n/n8n-nodes-langchain.outputParserStructured",
   173→        "typeVersion": 1.1,
   174→        "position": pos,

   176→        "name": name

   179→def ai_conn(source, target, conn_type):
   180→    return {
   181→        source: {
   182→            "ai_" + conn_type: [[{"node": target, "type": "ai_" + conn_type, "index": 0}]]
   183→        }
   184→    }
   185→
   186→def main_conn(source, target):
   187→    return {
   188→        source: {
   189→            "main": [[{"node": target, "type": "main", "index": 0}]]
   190→        }
   191→    }
   192→
   193→def merge_dicts(dicts):
   194→    """Merge connection dicts, appending lists for same connection types (e.g., ai_tool)."""
   195→    result = {}
   196→    for d in dicts:
   197→        for src, targets in d.items():
   198→            if src not in result:
   199→                result[src] = {}
   200→            for conn_type, conn_list in targets.items():
   201→                if conn_type in result[src]:
   202→                    # Append to existing list (for multiple ai_tool connections)
   203→                    result[src][conn_type].extend(conn_list)
   204→                else:
   205→                    result[src][conn_type] = list(conn_list)
   206→    return result

  1934→### 12.1 Error Categories
  1935→
  1936→| Category | HTTP Codes | Strategy |
  1937→|----------|-----------|----------|
  1938→| **Authentication** | 401, 403 | Refresh token, check credentials |
  1939→| **Rate Limit** | 429 | Exponential backoff, respect Retry-After |
  1940→| **Validation** | 400, 422 | Fix request parameters, log error |
  1941→| **Server** | 500, 502, 503 | Retry with backoff, fallback to alternative |
  1942→| **Timeout** | 408, 504 | Retry once, then alert |

  1960→## 13. Security Best Practices
  1961→
  1962→### 13.1 Credential Management
  1963→- Store all API keys and tokens in n8n Credentials (encrypted)
  1964→- Never hardcode credentials in workflow JSON
  1965→- Use environment variables for sensitive configuration
  1966→- Rotate API keys on a regular schedule (quarterly minimum)
  1967→- Use separate credentials for each environment
  1968→
  1969→### 13.2 Data Protection
  1970→- Encrypt sensitive data in transit (HTTPS/TLS)
  1971→- Mask PII in logs and error messages
  1972→- Implement data retention policies per platform
  1973→- Comply with GDPR, CCPA, and regional privacy laws
  1974→- Use webhook signature verification for all incoming data
  1975→
  1976→### 13.3 Access Control
  1977→- Use principle of least privilege for API permissions
  1978→- Implement role-based access in n8n
  1979→- Audit workflow execution logs regularly
  1980→- Set up IP allowlists for webhook endpoints
  1981→- Monitor for unusual API usage patterns
