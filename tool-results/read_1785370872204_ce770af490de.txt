   223→def merge_dicts(dicts):
   224→    """Merge multiple connection dicts."""
   225→    result = {}
   226→    for d in dicts:
   227→        for k, v in d.items():
   228→            if k in result:
   229→                for ck, cv in v.items():
   230→                    if ck in result[k]:
   231→                        result[k][ck].extend(cv)
   232→                    else:
   233→                        result[k][ck] = cv
   234→            else:
   235→                result[k] = v
   236→    return result
