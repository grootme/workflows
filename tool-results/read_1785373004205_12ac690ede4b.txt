   219→def merge_dicts(dicts):
   220→    result = {}
   221→    for d in dicts:
   222→        for k, v in d.items():
   223→            if k in result:
   224→                for ck, cv in v.items():
   225→                    if ck in result[k]:
   226→                        result[k][ck].extend(cv)
   227→                    else:
   228→                        result[k][ck] = cv
   229→            else:
   230→                result[k] = v
   231→    return result
